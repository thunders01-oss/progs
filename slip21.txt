Def dls(graph, node, goal, depth):     
If node == goal:         
Return True     
If depth == 0:         
Return False      
For child in graph[node]:         
If dls(graph, child, goal, depth – 1):             
Return True     
Return False  
Def ids(graph, start, goal, limit):    
For depth in range(limit + 1):         
Print(“Depth:”, depth)
