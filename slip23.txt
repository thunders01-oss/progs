Def f(x): 
Return -x**2 + 4*x X = 0 
While True: 
If f(x + 1) > f(x): 
X += 1 Elif f(x – 1) > f(x): 
X -= 1 
Else: 
Break 
Print(“Maximum value:”, f(x)) 
Print(“At x =”, x)
